#!/bin/bash
set -e

# Herramientas necesarias
dnf install -y curl git libcap

# Obtener última versión de code-server
VERSION=$(curl -fsSLI \
  -o /dev/null \
  -w '%{url_effective}' \
  https://github.com/coder/code-server/releases/latest \
  | sed 's|.*/v||')

# Descargar code-server
curl -fL \
  -o /tmp/code-server.rpm \
  "https://github.com/coder/code-server/releases/download/v${VERSION}/code-server-${VERSION}-amd64.rpm"

# Instalar
dnf install -y /tmp/code-server.rpm

# Permitir que Node pueda utilizar puertos <1024
setcap 'cap_net_bind_service=+ep' /usr/lib/code-server/lib/node

# Crear configuración
mkdir -p /home/ec2-user/.config/code-server

cat > /home/ec2-user/.config/code-server/config.yaml <<'EOF'
bind-addr: 0.0.0.0:80
auth: password
password: Password1234!
cert: false
EOF

# Permisos
chown -R ec2-user:ec2-user /home/ec2-user/.config

# Arrancar automáticamente
systemctl daemon-reload
systemctl enable --now code-server@ec2-user
